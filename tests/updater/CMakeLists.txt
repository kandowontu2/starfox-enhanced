cmake_minimum_required(VERSION 3.22)
# Build the real service without configuring unrelated graphics/Metal kernels.
get_filename_component(STARFOX_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
file(STRINGS "${STARFOX_ROOT}/CMakeLists.txt" version_line REGEX "^project\\(starfox_enhanced VERSION")
string(REGEX MATCH "VERSION ([0-9.]+)" version_match "${version_line}")
project(starfox_update_checks VERSION ${CMAKE_MATCH_1} LANGUAGES C CXX)
include(FetchContent)
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
set(SDL_UNIX_CONSOLE_BUILD ON CACHE BOOL "" FORCE)
foreach(subsystem VIDEO AUDIO GPU RENDER CAMERA JOYSTICK HAPTIC POWER SENSOR DIALOG TRAY)
    set(SDL_${subsystem} OFF CACHE BOOL "" FORCE)
endforeach()
FetchContent_Declare(sdl3
    URL https://www.libsdl.org/release/SDL3-3.4.14.tar.gz
    URL_HASH SHA256=30d4aa2b3037718142b32dffd4e72f917ebb6cc5227150e7bb9c45efb2153aeb)
FetchContent_MakeAvailable(sdl3)
include("${STARFOX_ROOT}/cmake/UpdateService.cmake")
enable_testing()
add_executable(starfox_update_tests "${STARFOX_ROOT}/tests/update_tests.cpp")
target_link_libraries(starfox_update_tests PRIVATE starfox_updates SDL3::SDL3)
add_test(NAME starfox_update_tests COMMAND starfox_update_tests)
if(WIN32 OR CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_executable(starfox_update_exit_fixture "${STARFOX_ROOT}/tests/update_exit_fixture.cpp")
endif()
